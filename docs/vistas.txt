CREATE   VIEW [dbo].[IMPLT_accounts]
AS
WITH IMPLT_accounts AS (
    SELECT
        cl.CLIENTE,
		cl.RUBRO4_CLIENTE,
        cl.NOMBRE,--Name1
        cl.ALIAS,--Name2
        RTRIM(LTRIM(REPLACE(REPLACE(cl.CONTRIBUYENTE, '|', ''), '-', ''))) AS NIF,
        cl.DIRECCION,--Direcion1
        cl.PAIS,
        cl.TELEFONO1,
        cl.FAX,
        cl.E_MAIL,
        cl.U_CLUSTER,
        cl.ACTIVO,
        cl.CATEGORIA_CLIENTE,
        cl.ZONA,
        cl.NIVEL_PRECIO,
        cl.U_LATITUD,
        cl.U_LONGITUD,
        cl.MONEDA,
        cl.U_607_CLIENTE_GENERICO,
        cl.EXENTO_IMPUESTOS,
        cl.CONTACTO,
        cl.VENDEDOR,
        z.NOMBRE AS ZONA_NOMBRE,
		CL.U_TRANSFER_STATUS, 
        CL.CONDICION_PAGO,
        CL.U_IS_SYNCS
    FROM CATELLI.CLIENTE cl
    INNER JOIN CATELLI.zona z ON cl.zona = z.zona
    WHERE cl.ACTIVO = 'S'
      AND cl.CLIENTE NOT LIKE 'N%'
      AND cl.VENDEDOR NOT IN ('999', '998', '22')
      AND cl.NIVEL_PRECIO NOT LIKE 'OTROS' --AND U_ESTATUS = 'Nuevo'
      and CL.U_IS_SYNCS like '1'
)
SELECT
    CASE
        WHEN SUBSTRING(CLIENTE, PATINDEX('%[A-Za-z]%', CLIENTE), 1) NOT LIKE 'O'
             THEN 'CN' + CLIENTE
        ELSE CLIENTE
    END AS Code_ofClient,
	  RUBRO4_CLIENTE AS Code_Account,
    CASE WHEN RTRIM(LTRIM(REPLACE(REPLACE(NOMBRE, '|', ''), ',', ''))) LIKE '' THEN 'NA' ELSE RTRIM(LTRIM(REPLACE(REPLACE(NOMBRE, '|', ''), ',', ''))) END AS Name1,
    CASE WHEN RTRIM(LTRIM(REPLACE(REPLACE(NOMBRE, '|', ''), ',', ''))) LIKE '' THEN 'NA' ELSE RTRIM(LTRIM(REPLACE(REPLACE(NOMBRE, '|', ''), ',', ''))) END AS Name2,
    NIF,
    PAIS AS Code_Country,
	CASE WHEN 
    SUBSTRING(
         LTRIM(REPLACE(REPLACE(REPLACE(CAST(DIRECCION AS VARCHAR(MAX)), '|', ''), CHAR(13), ''), CHAR(10), ''))
         , 1, 75) LIKE '' THEN 'NA' ELSE  SUBSTRING(
         LTRIM(REPLACE(REPLACE(REPLACE(CAST(DIRECCION AS VARCHAR(MAX)), '|', ''), CHAR(13), ''), CHAR(10), ''))
         , 1, 75) END  AS Address1,
		 CASE WHEN  SUBSTRING(
         LTRIM(REPLACE(REPLACE(REPLACE(CAST(DIRECCION AS VARCHAR(MAX)), '|', ''), CHAR(13), ''), CHAR(10), ''))
         , 76, 150) LIKE '' THEN 'NA' ELSE  SUBSTRING(
         LTRIM(REPLACE(REPLACE(REPLACE(CAST(DIRECCION AS VARCHAR(MAX)), '|', ''), CHAR(13), ''), CHAR(10), ''))
         , 76, 150) END AS Address2,
    CASE
        WHEN SUBSTRING(ZONA, PATINDEX('%[A-Za-z]%', ZONA), 1) NOT LIKE 'O'
             THEN 'CN' + ZONA
        ELSE ZONA
    END AS Code_Province,
    ZONA_NOMBRE AS Province,
    CASE WHEN SUBSTRING(LTRIM(REPLACE(TELEFONO1, '/', '')), 1, 12) LIKE '' THEN 'NA' ELSE SUBSTRING(LTRIM(REPLACE(TELEFONO1, '/', '')), 1, 12) END  AS Phone1,
    CASE WHEN SUBSTRING(LTRIM(REPLACE(TELEFONO1, '/', '')), 13, 30) LIKE '' THEN 'NA' ELSE SUBSTRING(LTRIM(REPLACE(TELEFONO1, '/', '')), 13, 30) END AS Phone2,
    CASE WHEN RTRIM(LTRIM(REPLACE(FAX, '/', ''))) LIKE '' THEN 'NA' ELSE  RTRIM(LTRIM(REPLACE(FAX, '/', ''))) END AS Fax,
    CASE WHEN E_MAIL LIKE '' THEN 'NA' ELSE E_MAIL END AS E_mail,
    '02' AS Code_Type,
    CASE
        WHEN U_CLUSTER IS NULL THEN 'N'
        ELSE U_CLUSTER
    END AS Code_Potential,
    CASE
        WHEN ACTIVO = 'S' THEN '1'
        ELSE '0'
    END AS Code_Status,
    CATEGORIA_CLIENTE AS Code_Agrupation1,
	CATEGORIA_CLIENTE AS Code_ClassificationA,
	Code_ClassificationB = CASE CATEGORIA_CLIENTE WHEN 'A1' THEN 'A1'
				WHEN 'A2' THEN 'A1'
				WHEN 'A3' THEN 'A1'
				WHEN 'Q1' THEN 'D1'
				WHEN 'D1' THEN 'D1'
				WHEN 'D2' THEN 'D1'
				WHEN 'SUR' THEN 'D1'
				WHEN 'C1' THEN 'C1'
				WHEN 'SM' THEN 'C1'
				WHEN 'C2' THEN 'C1'
				WHEN 'C3' THEN 'C1'
				WHEN 'B1' THEN 'OT'
				WHEN 'OT' THEN 'OT'
				WHEN 'EM' THEN 'OT'
				WHEN 'FA' THEN 'OT'
				WHEN 'CA' THEN 'OT'
				WHEN 'H1' THEN 'OT'
				else 'OT' END,
    NIVEL_PRECIO AS Code_Price_List,
    CASE WHEN U_LATITUD LIKE '' OR  U_LATITUD IS NULL THEN 'NA'  ELSE U_LATITUD  END  AS Latitude,
     CASE WHEN U_LONGITUD LIKE '' OR  U_LONGITUD IS NULL THEN 'NA'  ELSE U_LONGITUD  END    AS Length,
    MONEDA AS Code_Currency,
    --CASE
    --    WHEN NIVEL_PRECIO IN ('COLGATE', 'BRONZE', 'GOLD', 'SILVER', 'LISTA_MA', 'LISTA_INN', 'LISTA_SM', 'LISTA_IND', 'IND_VIP')
    --         THEN '01'
    --    ELSE NIVEL_PRECIO
    --END AS Code_Warehouse,
    '01' AS Code_Warehouse,
    CASE 
    WHEN CONDICION_PAGO >= 3 THEN '3' ELSE '2' END AS Code_Payment_Term,
    CASE
        WHEN U_607_CLIENTE_GENERICO = 'NO' THEN '0'
        ELSE '1'
    END AS Gen_Account, 
    CONTACTO AS Contact_Name,
    CONTACTO AS Owner,    
    CASE
        WHEN SUBSTRING(VENDEDOR, PATINDEX('%[A-Za-z]%', VENDEDOR), 1) NOT LIKE 'O'
             THEN 'C' + VENDEDOR
        ELSE VENDEDOR END AS Code_Seller,
	CASE 
		WHEN U_TRANSFER_STATUS = '1' THEN '1'
		WHEN U_TRANSFER_STATUS = '2' THEN '2'
		WHEN U_TRANSFER_STATUS = '3' THEN '3'
		else '0'
	end as Transfer_status
FROM IMPLT_accounts;

CREATE   VIEW [dbo].[IMPLT_accounts_array]
AS
WITH AccountsArray AS (
    SELECT 
        CASE 
            WHEN u_type = '19' THEN '1'
            WHEN u_descrip IN ('A', 'B', 'C', 'D', 'F', 'N', 'S') THEN u_descrip
            ELSE U_CODIGO
        END AS Code,
        u_type AS Type,
        u_descrip AS Description,
        1 AS Transfer_status
    FROM CATELLI.U_ACCOUNTS_ARRAY
    UNION ALL
   SELECT 
        Tipo AS Code,
        '32' AS Type,
        descripcion AS Description,
        1 AS Transfer_status
    FROM CATELLI.NCF_CONSECUTIVO
    WHERE ACTIVO LIKE 'S'
    UNION ALL
    SELECT 
        pais AS Code,
        '50' AS Type,
        nombre AS Description,
        1 AS Transfer_status
    FROM CATELLI.pais
    UNION ALL
    SELECT 
        nivel_precio AS Code,
        '53' AS Type,
        nivel_precio AS Description,
        1 AS Transfer_status
    FROM CATELLI.nivel_precio
    UNION ALL
    SELECT 
        moneda AS Code,
        '54' AS Type,
        nombre AS Description,
        1 AS Transfer_status
    FROM CATELLI.MONEDA
    
)
SELECT * FROM AccountsArray;

    SELECT
        pel.PEDIDO AS Order_Num,

        pel.PEDIDO AS Order_Num_ofClient,
        'S' AS Type_Rec,
        CONVERT(VARCHAR, pel.FECHA_ENTREGA, 112) AS Date_Delivery,
        CONVERT(VARCHAR, pe.FECHA_PEDIDO, 112) AS Order_Date,
        CASE
           WHEN SUBSTRING(pe.cliente, PATINDEX('%[A-Za-z]%', pe.cliente), 1) NOT LIKE 'O'
                 THEN 'CN' + pe.cliente
            ELSE pe.CLIENTE
        END AS Code_Account,

        ar.UNIDAD_VENTA AS Unit_Measure,
        CAST(pel.PRECIO_UNITARIO AS varchar(45)) AS Price_Br,
        CAST(CAST(pel.PRECIO_UNITARIO - (pel.MONTO_DESCUENTO / NULLIF(pel.CANTIDAD_PEDIDA, 0)) AS DECIMAL(18,2)) AS varchar(45)) AS Price,
        CAST(CAST((pel.PRECIO_UNITARIO - (pel.MONTO_DESCUENTO / NULLIF(pel.CANTIDAD_PEDIDA, 0))) * pel.CANTIDAD_PEDIDA AS DECIMAL(18,2)) AS varchar(45)) AS Total_Amount,

        CAST(CAST(((pel.PRECIO_UNITARIO - (pel.MONTO_DESCUENTO / NULLIF(pel.CANTIDAD_PEDIDA, 0))) * pel.CANTIDAD_PEDIDA) * (im.IMPUESTO1 / 100) AS DECIMAL(18,2)) AS varchar(45)) AS Amount_Tax1,
        'RD' AS Code_Currency,
        pe.CONDICION_PAGO AS Code_Paymentway,
            CASE 
                WHEN LEFT(pe.VENDEDOR, 1) BETWEEN 'A' AND 'Z' AND LEFT(pe.VENDEDOR, 1) <> 'O'
                     THEN 'C' + pe.VENDEDOR
                ELSE pe.VENDEDOR
            END AS Code_Seller,
            1 AS Transfer_Status
    FROM CATELLI.PEDIDO_LINEA pel
    INNER JOIN CATELLI.PEDIDO pe ON pe.PEDIDO = pel.PEDIDO
    INNER JOIN CATELLI.ARTICULO ar ON ar.ARTICULO = pel.ARTICULO
    INNER JOIN CATELLI.IMPUESTO im ON im.IMPUESTO = ar.IMPUESTO
    WHERE pel.CANTIDAD_PEDIDA <> 0
      AND pe.ESTADO = 'F' AND pe.RUBRO1 IS NULL--NOT LIKE 'PRIP'
      AND pe.VENDEDOR NOT IN ('999','998','22')
      AND pe.FECHA_PEDIDO >= DATEADD(MONTH, -3, CAST(GETDATE() AS DATE))
)

CREATE VIEW [dbo].[IMPLT_Orders]
as
WITH BaseData AS (
  SELECT 
    pl.PEDIDO,
    pl.PEDIDO_LINEA,
    pl.CANTIDAD_PEDIDA,
    pl.CANTIDAD_BONIFICAD,  -- Asegúrate que existe este campo
    pl.PRECIO_UNITARIO,
    pl.MONTO_DESCUENTO,
    pl.PORC_DESCUENTO,
    pl.PORC_IMPUESTO1,
    pl.PORC_IMPUESTO2,
    pE.RUBRO5,
    pe.FECHA_PROMETIDA,
    pe.FECHA_PEDIDO,
    pe.CLIENTE,
    pe.VENDEDOR,
    cl.detalle_direccion,
    ar.ARTICULO,
    ar.UNIDAD_ALMACEN,
    pl.BODEGA
  FROM CATELLI.PEDIDO_LINEA AS pl
  INNER JOIN CATELLI.PEDIDO AS pe ON pe.PEDIDO = pl.PEDIDO
  INNER JOIN CATELLI.CLIENTE AS cl ON cl.CLIENTE = pe.CLIENTE
  LEFT JOIN CATELLI.ARTICULO AS ar ON ar.ARTICULO = pl.ARTICULO  -- LEFT para no perder líneas
  WHERE pe.estado = 'N'
    -- REMOVÍ el filtro de cantidad_pedida para incluir todas las líneas
),
Calc AS (
  -- LÍNEAS DE CANTIDAD PEDIDA
  SELECT 
    PEDIDO,
    PEDIDO_LINEA,
    CAST(PEDIDO_LINEA AS VARCHAR(10)) + '-P' AS LINEA_TIPO,  -- ✅ CAST agregado
    'P' AS TIPO_LINEA,
    CANTIDAD_PEDIDA AS Cantidad,
    PRECIO_UNITARIO,
    MONTO_DESCUENTO,
    PORC_DESCUENTO,
    PORC_IMPUESTO1,
    PORC_IMPUESTO2,
    RUBRO5,
    FECHA_PROMETIDA,
    FECHA_PEDIDO,
    CLIENTE,
    VENDEDOR,
    detalle_direccion,
    ARTICULO,
    UNIDAD_ALMACEN,
    BODEGA,
    (CANTIDAD_PEDIDA * PRECIO_UNITARIO) AS SubTotal,
    ((CANTIDAD_PEDIDA * PRECIO_UNITARIO) - MONTO_DESCUENTO) AS TotalAmount
  FROM BaseData
  WHERE CANTIDAD_PEDIDA <> 0  -- Solo incluir si hay cantidad pedida

  UNION ALL

  -- LÍNEAS DE CANTIDAD BONIFICADA
  SELECT 
    PEDIDO,
    PEDIDO_LINEA,
    CAST(PEDIDO_LINEA AS VARCHAR(10)) + '-B' AS LINEA_TIPO,  -- ✅ CAST agregado
    'B' AS TIPO_LINEA,  -- Identificador adicional
    CANTIDAD_BONIFICAD AS Cantidad,
    0 AS PRECIO_UNITARIO,  -- Bonificación no tiene precio
    0 AS MONTO_DESCUENTO,
    0 AS PORC_DESCUENTO,
    0 AS PORC_IMPUESTO1,
    0 AS PORC_IMPUESTO2,
    RUBRO5,
    FECHA_PROMETIDA,
    FECHA_PEDIDO,
    CLIENTE,
    VENDEDOR,
    detalle_direccion,
    ARTICULO,
    UNIDAD_ALMACEN,
    BODEGA,
    0 AS SubTotal,
    0 AS TotalAmount
  FROM BaseData
  WHERE CANTIDAD_BONIFICAD > 0  -- Solo incluir si hay bonificación
)
SELECT
  'CATELLI' AS Code_Unit_Org,
  'CATELLI' AS Code_Sales_Org,
  PEDIDO AS Order_Num_ofClient,

CREATE   VIEW [dbo].[IMPLT_products_measure]
AS
WITH ProductMeasure AS (
    SELECT 
        ar.ARTICULO AS Code_Product,
        CASE 
            WHEN LEFT(ar.MANUFACTURADOR, 2) IN ('CA', 'FA', 'CO', 'GA', 'SA', 'DI', 'RI', 'UN', 'CB','BO','LI','SE','PR','BT')
                 THEN CASE LEFT(ar.MANUFACTURADOR, 2)
                         WHEN 'CA' THEN 'CAJA'
                         WHEN 'FA' THEN 'FDR'
                         WHEN 'CO' THEN 'CMB'
                         WHEN 'GA' THEN 'GLN'
                         WHEN 'SA' THEN 'SACO'
                         WHEN 'DI' THEN 'DSP'
                         WHEN 'RI' THEN 'RST'
                         WHEN 'UN' THEN 'UND'
                         WHEN 'CB' THEN 'CB'
						 WHEN 'MA' THEN 'MUE'
						 WHEN 'PR' THEN 'PRO'
						 WHEN 'LI' THEN 'LIQ'
						 WHEN 'BO' THEN 'BOT'
						 WHEN 'SE' THEN 'SER'
						 WHEN 'BT' THEN 'BUL'
                      END
            ELSE 'UND'
        END AS Unit_Measure,
        CAST(ar.PESO_BRUTO AS varchar(45)) AS Factor_Conversion, 
         CASE 
		WHEN ar.U_ESTATUS = 'Nuevo' THEN '1'
		WHEN ar.U_ESTATUS = 'EnEspera' THEN '3'
		else '2'
	end as Transfer_status
    FROM CATELLI.ARTICULO ar
    INNER JOIN CATELLI.UNIDAD_DE_MEDIDA un ON ar.UNIDAD_VENTA = un.UNIDAD_MEDIDA
    WHERE ar.CLASIFICACION_4 NOT IN ('GND')
      AND ar.TIPO NOT IN ('K')
      AND LEFT(ar.MANUFACTURADOR, 2) NOT IN ('UN', 'DI') AND AR.ACTIVO  = 'S'
    UNION ALL
    SELECT 
        ar.ARTICULO AS Code_Product,
        ar.UNIDAD_VENTA AS Unit_Measure,
        CAST(ar.FACTOR_CONVER_1 AS varchar(45)) AS Factor_Conversion,
          CASE 
		WHEN ar.U_ESTATUS = 'Nuevo' THEN '1'
		WHEN ar.U_ESTATUS = 'EnEspera' THEN '3'
		else '2'
	end as Transfer_status 
    FROM CATELLI.ARTICULO ar
    INNER JOIN CATELLI.UNIDAD_DE_MEDIDA un ON ar.UNIDAD_VENTA = un.UNIDAD_MEDIDA
    WHERE  ar.CLASIFICACION_4 NOT IN ('GND')
      AND ar.TIPO NOT IN ('K')
      AND NOT (
          (LEFT(ar.MANUFACTURADOR, 2) IN ('CO', 'CA', 'FA', 'GA', 'SA', 'FA', 'CB') AND ar.PESO_BRUTO = 1)

AS
WITH ProductOrganization AS (
    SELECT 
        ar.ARTICULO AS Code_Product,
        'CATELLI' AS Code_Unit_Org,
        'CATELLI' AS Code_Sales_Org,
        CASE WHEN ar.ACTIVO = 'S' THEN '1' ELSE '0' END AS Code_Status,
        1 AS Unit_Box,
        CASE 
            WHEN ar.MANUFACTURADOR LIKE 'CA%' AND ar.PESO_BRUTO = 1 THEN 'CAJA'
            WHEN ar.MANUFACTURADOR LIKE 'FA%' AND ar.PESO_BRUTO = 1 THEN 'FDR'
            WHEN ar.MANUFACTURADOR LIKE 'CO%' AND ar.PESO_BRUTO = 1 THEN 'CMB'
            WHEN ar.MANUFACTURADOR LIKE 'GA%' AND ar.PESO_BRUTO = 1 THEN 'GLN'
            WHEN ar.MANUFACTURADOR LIKE 'SA%' AND ar.PESO_BRUTO = 1 THEN 'SACO'
            WHEN ar.MANUFACTURADOR LIKE 'DI%' AND ar.PESO_BRUTO = 1 THEN 'DSP'
            WHEN ar.MANUFACTURADOR LIKE 'RI%' AND ar.PESO_BRUTO = 1 THEN 'RST'
            WHEN ar.MANUFACTURADOR LIKE 'UN%' AND ar.PESO_BRUTO = 1 THEN 'UND'
            ELSE ar.UNIDAD_VENTA 
        END AS Unit_Type_Sales,
        CASE 
            WHEN ar.UNIDAD_EMPAQUE LIKE 'CA%' THEN 'CAJA'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'FA%' THEN 'FDR'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'CO%' THEN 'CMB'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'GA%' THEN 'GLN'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'SA%' THEN 'SACO'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'DI%' THEN 'DSP'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'RI%' THEN 'RST'
            WHEN ar.UNIDAD_EMPAQUE LIKE 'UN%' THEN 'UND'
            ELSE ar.UNIDAD_EMPAQUE 
        END AS Unit_Type_Inv,
        CAST(ar.PESO_BRUTO AS varchar(45)) AS Factor_Conversion,
        '0' AS Trazability,
        CASE 
		WHEN ar.U_ESTATUS = 'Nuevo' THEN '1'
		WHEN ar.U_ESTATUS = 'EnEspera' THEN '3'
		else '2'
	end as Transfer_status
    FROM CATELLI.ARTICULO ar
    WHERE ar.CLASIFICACION_4 NOT LIKE 'GND%' --AND ar.ACTIVO = 'S'
   
)
SELECT * FROM ProductOrganization;

    1 AS Transfer_status, 
    LEFT(c.mes_anno, 7) AS ANNO_MES,
    c.mes
FROM dbo.cuota c
LEFT JOIN dbo.Ventas_Marca_Mensual vm 
    ON c.Subcategoria = vm.Code_Category 
    AND c.vendedor = vm.Code_Seller
    AND LEFT(c.mes_anno, 7) = vm.ANNO_MES;
